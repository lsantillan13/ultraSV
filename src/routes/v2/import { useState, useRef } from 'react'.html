import { useState, useRef } from 'react'; export default function VoxReader({ slug }) { const [playing, setPlaying] = useState(false); const [loading, setLoading] = useState(false); const audioRef = useRef(null); const handlePlay = async () => { const API = import.meta.env.VITE_API_URL || "http://localhost:8080"; setLoading(true); try { // 1. intenta obtener audio existente let res = await fetch(`${API}/api/v2/tts/${slug}`).then(r=>r.json()); let audioUrl = res.data?.audioUrl; // 2. si no existe, lo genera if (!audioUrl) { const gen = await fetch(`${API}/api/v2/tts/${slug}/generate`, { method: 'POST' }).then(r=>r.json()); audioUrl = gen.url; } if (audioUrl) { // audio premium if (!audioRef.current) { audioRef.current = new Audio(`${API}${audioUrl}`); audioRef.current.onended = () => setPlaying(false); } if (playing) { audioRef.current.pause(); setPlaying(false); } else { audioRef.current.play(); setPlaying(true); } } else { // fallback: Web Speech API con texto limpio (sin HTML) const plain = res.data?.plain || res.data?.resume; const utter = new SpeechSynthesisUtterance(plain); utter.lang = 'es-AR'; utter.rate = 1.05; if (playing) { speechSynthesis.cancel(); setPlaying(false); } else { speechSynthesis.speak(utter); setPlaying(true); utter.onend = () => setPlaying(false); } } } catch(e){ console.error(e); } setLoading(false); }; return ( ); }
